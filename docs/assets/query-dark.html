SELECT p.name, SUM(li.amount) / 100.0 AS revenue FROM invoice_line_items li JOIN invoices i ON i.id = li.invoice JOIN prices pr ON pr.id = json_extract(li.pricing, '$.price_details.price') JOIN products p ON p.id = pr.product WHERE i.status = 'paid' AND i.created > unixepoch('now', '-90 days') GROUP BY p.name ORDER BY revenue DESC LIMIT 5;
